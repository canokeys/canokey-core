cmake_minimum_required(VERSION 3.20)
project(canokey-core C)
set(CANOKEY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FindPkgConfig)
include(CTest)
include(cmake/ReleaseVersions.cmake)
option(ENABLE_TESTS "Enable native helper tests and host sanitizers" OFF)
option(ENABLE_APDU_REPLAY "Build the correctness APDU replay tool" OFF)
option(CANOKEY_FUZZING "Build the libFuzzer harness for the Rust core (Clang, dedicated build directory)" OFF)
option(CTAP_RESTRICT_ALGORITHMS "Restrict FIDO credential algorithms to ES256 and Ed25519" OFF)
include(cmake/Profiles.cmake)
include(cmake/Cargo.cmake)
if(ENABLE_TESTS)
  add_compile_options(-Wall --coverage -fsanitize=address -fsanitize=undefined)
  add_compile_options($<$<CONFIG:Debug>:-fno-omit-frame-pointer>)
  add_link_options(--coverage -fsanitize=address -fsanitize=undefined)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    add_compile_options(-fstack-usage)
  endif()
  list(APPEND CMAKE_MODULE_PATH "${CANOKEY_ROOT}/cmake/Modules")
  find_package(CMocka CONFIG REQUIRED)
  include(AddCMockaTest)
  include(AddMockedTest)
endif()
if(CANOKEY_FUZZING)
  # Instrument every C object for libFuzzer coverage and ASan/UBSan. All
  # executables in this build tree must link the sanitizer runtimes; use a
  # dedicated build directory rather than mixing with other host tools.
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "CANOKEY_FUZZING requires Clang (libFuzzer entrypoint)")
  endif()
  add_compile_options(-fsanitize=fuzzer-no-link -fsanitize=address -fsanitize=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address -fsanitize=undefined)
endif()
add_rust_host_archive(canokey-rust-core rust-core-build cargo "${FEATURES}")
add_subdirectory(native)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(CANOKEY_VIRTUAL_CARD OR CANOKEY_APDU_REPLAY OR CANOKEY_FUZZING)
  add_subdirectory(crates/host)
endif()
